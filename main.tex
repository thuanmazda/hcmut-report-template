def calculate_factorial(n):
    """Ham tinh giai thua cua mot so nguyen duong"""
    if n == 0 or n == 1:
        return 1
    else:
        return n * calculate_factorial(n - 1)

# Tinh va in ket qua ra man hinh
number = 5
print(f"Giai thua cua {number} la: {calculate_factorial(number)}")

    input clk,
    input reset,
    input d,
    output reg q
);

    // Kich hoat tai suon len cua xung nhip clk
    always @(posedge clk) begin
        if (reset)
            q <= 1'b0;
        else
            q <= d;
    end

endmodule
